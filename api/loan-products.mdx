---
title: "Loan Products API"
description: "Configure and manage loan product templates"
---

The Loan Products API allows you to create and manage loan product configurations that define the terms, interest rates, fees, and other parameters for different types of loans offered by your institution.

## Get All Loan Products

<Expandable title="GET /api/loan-products">
  Retrieve all loan products with pagination.

  <ParamField query="page" type="integer">
    Page number for pagination (0-indexed)
  </ParamField>

  <ParamField query="size" type="integer">
    Number of results per page
  </ParamField>

  <ParamField query="show_all" type="boolean" default="false">
    If true, returns all products including inactive ones. If false, returns only active products.
  </ParamField>

  **Response**

  <ResponseField name="content" type="array">
    Array of loan product objects
    
    <Expandable title="Loan Product Object">
      <ResponseField name="id" type="long">
        Loan product identifier
      </ResponseField>

      <ResponseField name="name" type="string">
        Product name
      </ResponseField>

      <ResponseField name="code" type="string">
        Unique product code
      </ResponseField>

      <ResponseField name="status" type="string">
        Product status (ACTIVE or INACTIVE)
      </ResponseField>

      <ResponseField name="currencyId" type="long">
        Currency identifier
      </ResponseField>

      <ResponseField name="hasPayees" type="boolean">
        Whether the product supports payees (disbursement to multiple parties)
      </ResponseField>

      <ResponseField name="amountMin" type="decimal">
        Minimum loan amount
      </ResponseField>

      <ResponseField name="amountMax" type="decimal">
        Maximum loan amount
      </ResponseField>

      <ResponseField name="interestRateMin" type="decimal">
        Minimum interest rate
      </ResponseField>

      <ResponseField name="interestRateMax" type="decimal">
        Maximum interest rate
      </ResponseField>

      <ResponseField name="maturityMin" type="integer">
        Minimum loan term
      </ResponseField>

      <ResponseField name="maturityMax" type="integer">
        Maximum loan term
      </ResponseField>

      <ResponseField name="scheduleType" type="string">
        Default schedule type (ANNUITY, FLAT, DECLINING_BALANCE, etc.)
      </ResponseField>

      <ResponseField name="availability" type="array">
        Array of profile types this product is available for (PERSON, COMPANY, GROUP)
      </ResponseField>

      <ResponseField name="provisioning" type="array">
        Provisioning rules for the product
      </ResponseField>
    </Expandable>
  </ResponseField>

  <ResponseField name="totalElements" type="long">
    Total number of products
  </ResponseField>

  <ResponseField name="totalPages" type="integer">
    Total number of pages
  </ResponseField>

  <CodeGroup>
  ```bash cURL
  curl -X GET "https://api.opencbs.com/api/loan-products?page=0&size=20" \
    -H "Authorization: Bearer YOUR_ACCESS_TOKEN"
  ```

  ```javascript JavaScript
  const response = await fetch('https://api.opencbs.com/api/loan-products?page=0&size=20', {
    headers: {
      'Authorization': 'Bearer YOUR_ACCESS_TOKEN'
    }
  });
  const products = await response.json();
  ```

  ```python Python
  import requests

  response = requests.get(
    'https://api.opencbs.com/api/loan-products',
    params={'page': 0, 'size': 20},
    headers={'Authorization': 'Bearer YOUR_ACCESS_TOKEN'}
  )
  products = response.json()
  ```
  </CodeGroup>
</Expandable>

## Create Loan Product

<Expandable title="POST /api/loan-products">
  Create a new loan product. This creates a maker-checker request that requires approval.

  **Request Body**

  <ParamField body="name" type="string" required>
    Product name
  </ParamField>

  <ParamField body="code" type="string" required>
    Unique product code
  </ParamField>

  <ParamField body="currencyId" type="long" required>
    Currency identifier
  </ParamField>

  <ParamField body="amountMin" type="decimal" required>
    Minimum loan amount
  </ParamField>

  <ParamField body="amountMax" type="decimal" required>
    Maximum loan amount
  </ParamField>

  <ParamField body="interestRateMin" type="decimal" required>
    Minimum interest rate
  </ParamField>

  <ParamField body="interestRateMax" type="decimal" required>
    Maximum interest rate
  </ParamField>

  <ParamField body="maturityMin" type="integer" required>
    Minimum loan term
  </ParamField>

  <ParamField body="maturityMax" type="integer" required>
    Maximum loan term
  </ParamField>

  <ParamField body="gracePeriodMin" type="integer">
    Minimum grace period (default: 0)
  </ParamField>

  <ParamField body="gracePeriodMax" type="integer">
    Maximum grace period (default: 0)
  </ParamField>

  <ParamField body="scheduleType" type="string" required>
    Schedule generation type
  </ParamField>

  <ParamField body="scheduleBasedType" type="string">
    DAILY_INSTALLMENT or MONTHLY_INSTALLMENT
  </ParamField>

  <ParamField body="hasPayees" type="boolean">
    Enable multiple payees
  </ParamField>

  <ParamField body="fees" type="array">
    Array of fee IDs to associate
  </ParamField>

  <ParamField body="penalties" type="array">
    Array of penalty IDs to associate
  </ParamField>

  <ParamField body="accountList" type="object" required>
    Chart of accounts mapping
  </ParamField>

  <ParamField body="topUpAllow" type="boolean">
    Allow top-up loans
  </ParamField>

  <ParamField body="earlyPartialRepaymentFeeType" type="string">
    Early partial repayment fee type
  </ParamField>

  <ParamField body="earlyPartialRepaymentFeeValue" type="decimal">
    Early partial repayment fee value
  </ParamField>

  <ParamField body="availability" type="array" required>
    Profile types (e.g., ["PERSON", "COMPANY"])
  </ParamField>

  <ParamField body="provisioning" type="array">
    Provisioning rules
  </ParamField>

  **Response**

  <ResponseField name="id" type="long">
    Maker-checker request ID (requires approval before product is created)
  </ResponseField>

  <CodeGroup>


  ```javascript JavaScript
    headers: {
      'Authorization': 'Bearer YOUR_ACCESS_TOKEN',

    },
  ```
